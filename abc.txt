import tensorflow as tf
import numpy as np
import matplotlib.pyplot as plt

from tensorflow import keras
from tensorflow.keras import layers
print("TensorFlow Version:", tf.__version__)

(x_train, y_train), (x_test, y_test) = keras.datasets.mnist.load_data()

print("Training images:", x_train.shape)
print("Training labels:", y_train.shape)
print("Testing images:", x_test.shape)
print("Testing labels:", y_test.shape)

x_train = x_train / 255.0
x_test = x_test / 255.0

plt.imshow(x_train[0], cmap='gray')
plt.title("MNIST Handwritten Digit")
plt.axis('off')
plt.show()
print("Actual Digit:", y_train[0])

model = keras.Sequential([
    layers.Flatten(input_shape=(28, 28)),
    layers.Dense(128, activation='relu'),
    layers.Dense(64, activation='relu'),
    layers.Dense(10, activation='softmax')
])

model.summary()

model.compile(
    optimizer=keras.optimizers.SGD(learning_rate=0.01),
    loss='sparse_categorical_crossentropy',
    metrics=['accuracy']
)

history = model.fit(
    x_train,
    y_train,
    epochs=5,
    batch_size=32,
    validation_split=0.1
)

test_loss, test_accuracy = model.evaluate(x_test, y_test)
print("Test Loss:", test_loss * 100, "%")
print("Test Accuracy:", test_accuracy * 100, "%")

plt.plot(history.history['loss'], label='Training Loss')
plt.plot(history.history['val_loss'], label='Validation Loss')
plt.title("Training and Validation Loss")
plt.xlabel("Epochs")
plt.ylabel("Loss")
plt.legend()
plt.show()

plt.plot(history.history['accuracy'], label='Training Accuracy')
plt.plot(history.history['val_accuracy'], label='Validation Accuracy')
plt.title("Training and Validation Accuracy")
plt.xlabel("Epochs")
plt.ylabel("Accuracy")
plt.legend()
plt.show()

prediction = model.predict(x_test[:1])
predicted_digit = np.argmax(prediction)
plt.imshow(x_test[0], cmap='gray')
plt.title("Predicted Digit: " + str(predicted_digit))
plt.axis('off')
plt.show()
print("Actual Digit:", y_test[0])
print("Predicted Digit:", predicted_digit)

import numpy as np
import matplotlib.pyplot as plt
import tensorflow as tf
from tensorflow import keras
from tensorflow.keras import layers

(x_train, y_train), (x_test, y_test) = keras.datasets.mnist.load_data()
print("Training data:", x_train.shape)
print("Testing data:", x_test.shape)

x_train = x_train.astype('float32') / 255.0
x_test = x_test.astype('float32') / 255.0


x_train_normal = x_train[y_train == 0]

x_test_normal = x_test[y_test == 0]
x_test_anomaly = x_test[y_test != 0]
print("Normal training images:", x_train_normal.shape)
print("Normal testing images:", x_test_normal.shape)
print("Anomaly testing images:", x_test_anomaly.shape)


plt.figure(figsize=(10, 3))

for i in range(5):
    plt.subplot(1, 5, i + 1)
    plt.imshow(x_train_normal[i], cmap='gray')
    plt.axis('off')

plt.show()



input_img = keras.Input(shape=(28, 28))

x = layers.Flatten()(input_img)
x = layers.Dense(64, activation='relu')(x)
x = layers.Dense(32, activation='relu')(x)

encoded = layers.Dense(16, activation='relu')(x)

encoder = keras.Model(input_img, encoded)


latent_input = keras.Input(shape=(16,))

x = layers.Dense(32, activation='relu')(latent_input)
x = layers.Dense(64, activation='relu')(x)
x = layers.Dense(784, activation='sigmoid')(x)

decoded = layers.Reshape((28, 28))(x)
decoder = keras.Model(latent_input, decoded)

autoencoder_input = keras.Input(shape=(28, 28))

encoded_image = encoder(autoencoder_input)
decoded_image = decoder(encoded_image)

autoencoder = keras.Model(autoencoder_input, decoded_image)


autoencoder.compile(
    optimizer='adam',
    loss='mse',
    metrics=['mae']
)

history = autoencoder.fit(
    x_train_normal,
    x_train_normal,
    epochs=10,
    batch_size=128,
    validation_split=0.1
)


plt.plot(history.history['loss'], label='Training Loss')
plt.plot(history.history['val_loss'], label='Validation Loss')

plt.title('Autoencoder Training Loss')
plt.xlabel('Epochs')
plt.ylabel('Loss')
plt.legend()
plt.show()


train_predictions = autoencoder.predict(x_train_normal)

train_error = np.mean(
    np.square(x_train_normal - train_predictions),
    axis=(1, 2)
)

threshold = np.percentile(train_error, 95)

print("Anomaly Threshold:", threshold)


test_images = np.concatenate([
    x_test_normal[:100],
    x_test_anomaly[:100]
])

test_predictions = autoencoder.predict(test_images)

test_error = np.mean(
    np.square(test_images - test_predictions),
    axis=(1, 2)
)

results = test_error > threshold

print("Total Images:", len(test_images))
print("Detected Anomalies:", np.sum(results))
print("Detected Normal Images:", np.sum(~results))

plt.figure(figsize=(10, 4))

for i in range(5):
    plt.subplot(2, 5, i + 1)
    plt.imshow(test_images[i], cmap='gray')
    plt.title("Original")
    plt.axis('off')

    plt.subplot(2, 5, i + 6)
    plt.imshow(test_predictions[i], cmap='gray')
    plt.title("Reconstructed")
    plt.axis('off')

plt.show()
